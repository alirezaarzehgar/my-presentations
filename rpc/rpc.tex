\begin{frame}{What is Remote Procedure Call?}
  % A person who shouted to other with a call
  \begin{itemize}
      \pitem A way to execute a \textbf{procedure} that could be running in a separate application on the same
      machine or even potentially on the network.
      \pitem A client-server mechanism for \textbf{interprocess communication (IPC)}.
      \pitem Needed an specific protocol for client-server communication called \textbf{interface definition}.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Simple Golang RPC interface definition}
  \begin{minted}{go}
type Hello interface {
  Print(*PrintArgs, *PrintReply) error
}

type PrintArgs struct{}

type PrintReply struct {
  Msg string
}
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Simple Golang RPC Server}
  \begin{minted}{go}
type Hello struct{}

func (Hello) Print(a *PrintArgs, r *PrintReply) error {
  r.Msg = "Hello, World!"
  return nil
}

func main() {
  rpc.Register(new(Hello))

  l, _ := net.Listen("tcp", ":8888")
  for {
    conn, _ := l.Accept()
    rpc.ServeConn(conn)
  }
}
  \end{minted}
\end{frame}

\begin{frame}[fragile]{Simple Golang RPC Client}
  \begin{minted}{go}
func main() {
  c, _ := rpc.Dial("tcp", "localhost:8888")

  var reply PrintReply
  c.Call("Hello.Print", PrintArgs{}, &reply)

  fmt.Println(reply.Msg)
}

// OUTPUT: Hello, World!
  \end{minted}
\end{frame}

\begin{frame}[fragile]{gRPC Server}
  \begin{minted}[fontsize=\footnotesize]{go}
type hello struct {
  UnimplementedHelloServer
}

func (hello) Print(c context.Context, r *Request) (*Response, error) {
  return &Response{Message: "Hello, Wrold"}, nil
}

func main() {
  s := grpc.NewServer()
  RegisterHelloServer(s, hello{})

  l, _ := net.Listen("tcp", ":8888")
  s.Serve(l)
}
  \end{minted}
\end{frame}

\begin{frame}[fragile]{gRPC Client}
  \begin{minted}{go}
func main() {
  conn, _ := grpc.Dial(":8888", grpc.WithInsecure())
  client := NewHelloClient(conn)
  r, _ := client.Print(context.Background(), &Request{})

  fmt.Println(r.Message)
}

// OUTPUT: Hello, World!
  \end{minted}
\end{frame}
