	\begin{frame}[fragile]{Modifying effective IDs}
	\begin{itemize}
	    \pitem int setuid(uid\_t uid);
	    \begin{itemize}
		\pitem root: changes real, effective, and saved UID (all three)
		\pitem non-root: changes ONLY effective UID (to real OR saved UID)
	    \end{itemize}
	    \pitem int setgid(gid\_t gid);
	    \begin{itemize}
		\pitem root: changes real, effective, and saved GID (all three)
		\pitem non-root: changes ONLY effective GID (to real OR saved GID)
	    \end{itemize}
	    \pitem int seteuid(uid\_t euid);
	    \pitem int setegid(gid\_t egid);
	    \pitem int setreuid(uid\_t ruid, uid\_t euid);
	    \pitem int setregid(gid\_t rgid, gid\_t egid);
	\end{itemize}
	\end{frame}
